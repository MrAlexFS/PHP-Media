/**
 * РСЃРєР»СЋС‡РµРЅРёРµ РґР»СЏ РѕС€РёР±РѕРє РїСЂРѕРІРµСЂРєРё Р·Р°РіСЂСѓР¶Р°РµРјС‹С… С„Р°Р№Р»РѕРІ.
 */
final class UploadValidationException extends RuntimeException
{
}

/**
 * РџСЂРѕРІРµСЂСЏРµС‚ СЃРѕСЃС‚РѕСЏРЅРёРµ Р·Р°РіСЂСѓР¶РµРЅРЅРѕРіРѕ С„Р°Р№Р»Р° Рё РµРіРѕ СЂР°Р·РјРµСЂ.
 */
function validateIncomingUpload(
    array $file,
    int $maxBytes,
    string $label
): array {
    if (
        !isset($file['error'], $file['tmp_name'], $file['name'])
        || !is_string($file['tmp_name'])
        || !is_string($file['name'])
    ) {
        throw new UploadValidationException(
            'РќРµРєРѕСЂСЂРµРєС‚РЅС‹Рµ РґР°РЅРЅС‹Рµ Р·Р°РіСЂСѓР¶Р°РµРјРѕРіРѕ С„Р°Р№Р»Р°.'
        );
    }

    $errorCode = (int) $file['error'];

    if ($errorCode !== UPLOAD_ERR_OK) {
        $message = match ($errorCode) {
            UPLOAD_ERR_INI_SIZE,
            UPLOAD_ERR_FORM_SIZE =>
                "Р¤Р°Р№Р» {$label} РїСЂРµРІС‹С€Р°РµС‚ РґРѕРїСѓСЃС‚РёРјС‹Р№ СЂР°Р·РјРµСЂ.",

            UPLOAD_ERR_PARTIAL =>
                "Р¤Р°Р№Р» {$label} Р·Р°РіСЂСѓР¶РµРЅ РЅРµ РїРѕР»РЅРѕСЃС‚СЊСЋ.",

            UPLOAD_ERR_NO_TMP_DIR =>
                'РќР° СЃРµСЂРІРµСЂРµ РѕС‚СЃСѓС‚СЃС‚РІСѓРµС‚ РІСЂРµРјРµРЅРЅР°СЏ РґРёСЂРµРєС‚РѕСЂРёСЏ.',

            UPLOAD_ERR_CANT_WRITE =>
                'РќРµ СѓРґР°Р»РѕСЃСЊ Р·Р°РїРёСЃР°С‚СЊ С„Р°Р№Р» РЅР° СЃРµСЂРІРµСЂРµ.',

            default =>
                "РћС€РёР±РєР° Р·Р°РіСЂСѓР·РєРё С„Р°Р№Р»Р° {$label}.",
        };

        throw new UploadValidationException($message);
    }

    // РќРµ РїСЂРёРЅРёРјР°РµРј РїСЂРѕРёР·РІРѕР»СЊРЅС‹Р№ РїСѓС‚СЊ Рє Р»РѕРєР°Р»СЊРЅРѕРјСѓ С„Р°Р№Р»Сѓ.
    if (!is_uploaded_file($file['tmp_name'])) {
        throw new UploadValidationException(
            'Р¤Р°Р№Р» РЅРµ РїСЂРѕС€С‘Р» РїСЂРѕРІРµСЂРєСѓ РёСЃС‚РѕС‡РЅРёРєР° Р·Р°РіСЂСѓР·РєРё.'
        );
    }

    $actualSize = filesize($file['tmp_name']);

    if (
        $actualSize === false
        || $actualSize < 1
        || $actualSize > $maxBytes
        || (int) ($file['size'] ?? 0) > $maxBytes
    ) {
        throw new UploadValidationException(
            "РќРµРґРѕРїСѓСЃС‚РёРјС‹Р№ СЂР°Р·РјРµСЂ С„Р°Р№Р»Р° {$label}."
        );
    }

    return [
        'tmp_name' => $file['tmp_name'],
        'original_name' => $file['name'],
        'size_bytes' => (int) $actualSize,
    ];
}

/**
 * РЎРѕР·РґР°С‘С‚ Р·Р°РєСЂС‹С‚С‹Р№ РєР°С‚Р°Р»РѕРі, РµСЃР»Рё РѕРЅ РµС‰С‘ РЅРµ СЃСѓС‰РµСЃС‚РІСѓРµС‚.
 */
function ensurePrivateDirectory(string $directory): void
{
    if (
        !is_dir($directory)
        && !mkdir($directory, 0750, true)
        && !is_dir($directory)
    ) {
        throw new RuntimeException(
            'РќРµ СѓРґР°Р»РѕСЃСЊ СЃРѕР·РґР°С‚СЊ РєР°С‚Р°Р»РѕРі С„Р°Р№Р»РѕРІРѕРіРѕ С…СЂР°РЅРёР»РёС‰Р°.'
        );
    }

    if (!is_writable($directory)) {
        throw new RuntimeException(
            'РЈ РІРµР±-СЃРµСЂРІРµСЂР° РЅРµС‚ РїСЂР°РІ РЅР° Р·Р°РїРёСЃСЊ РІ С…СЂР°РЅРёР»РёС‰Рµ.'
        );
    }
}

/**
 * РџСЂРѕРІРµСЂСЏРµС‚ MIME-С‚РёРї С„Р°Р№Р»Р°.
 */
function detectMimeType(string $tmpName): string
{
    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mime = $finfo->file($tmpName);

    if (!is_string($mime) || $mime === '') {
        throw new UploadValidationException(
            'РќРµ СѓРґР°Р»РѕСЃСЊ РѕРїСЂРµРґРµР»РёС‚СЊ MIME-С‚РёРї С„Р°Р№Р»Р°.'
        );
    }

    return $mime;
}

/**
 * РЎРѕС…СЂР°РЅСЏРµС‚ Р°СѓРґРёРѕ РІ РїСЂРёРІР°С‚РЅРѕРј РєР°С‚Р°Р»РѕРіРµ.
 *
 * Р Р°Р·СЂРµС€С‘РЅРЅС‹Рµ СЂР°СЃС€РёСЂРµРЅРёСЏ: mp3, wav, ogg.
 * РњР°РєСЃРёРјР°Р»СЊРЅС‹Р№ СЂР°Р·РјРµСЂ: 50 РњРёР‘.
 */
function storeAudioUpload(?array $file): array
{
    if ($file === null) {
        throw new UploadValidationException(
            'РќРµРѕР±С…РѕРґРёРјРѕ РІС‹Р±СЂР°С‚СЊ Р°СѓРґРёРѕС„Р°Р№Р».'
        );
    }

    $upload = validateIncomingUpload(
        $file,
        50 * 1024 * 1024,
        'Р°СѓРґРёРѕ'
    );

    $extension = strtolower(
        pathinfo(
            $upload['original_name'],
            PATHINFO_EXTENSION
        )
    );

    // Р Р°СЃС€РёСЂРµРЅРёРµ СЃРѕРїРѕСЃС‚Р°РІР»СЏРµС‚СЃСЏ СЃ РґРѕРїСѓСЃС‚РёРјС‹РјРё MIME-С‚РёРїР°РјРё.
    $allowed = [
        'mp3' => [
            'audio/mpeg',
            'audio/mp3',
            'audio/x-mpeg',
        ],

        'wav' => [
            'audio/wav',
            'audio/x-wav',
            'audio/wave',
            'audio/vnd.wave',
        ],

        'ogg' => [
            'audio/ogg',
            'application/ogg',
        ],
    ];

    if (!isset($allowed[$extension])) {
        throw new UploadValidationException(
            'Р Р°Р·СЂРµС€РµРЅС‹ С‚РѕР»СЊРєРѕ С„Р°Р№Р»С‹ MP3, WAV Рё OGG.'
        );
    }

    $detectedMime = detectMimeType($upload['tmp_name']);

    if (!in_array(
        $detectedMime,
        $allowed[$extension],
        true
    )) {
        throw new UploadValidationException(
            'Р Р°СЃС€РёСЂРµРЅРёРµ С„Р°Р№Р»Р° РЅРµ СЃРѕРѕС‚РІРµС‚СЃС‚РІСѓРµС‚ РµРіРѕ MIME-С‚РёРїСѓ.'
        );
    }

    ensurePrivateDirectory(AUDIO_STORAGE_DIR);

    // РСЃС…РѕРґРЅРѕРµ РёРјСЏ РїРѕР»СЊР·РѕРІР°С‚РµР»СЏ РЅРµ РёСЃРїРѕР»СЊР·СѓРµС‚СЃСЏ РІ РїСѓС‚Рё.
    $filename = bin2hex(random_bytes(16))
        . '.'
        . $extension;

    $target = AUDIO_STORAGE_DIR
        . DIRECTORY_SEPARATOR
        . $filename;

    if (!move_uploaded_file(
        $upload['tmp_name'],
        $target
    )) {
        throw new RuntimeException(
            'РќРµ СѓРґР°Р»РѕСЃСЊ СЃРѕС…СЂР°РЅРёС‚СЊ Р°СѓРґРёРѕС„Р°Р№Р».'
        );
    }

    // Р”РѕРїРѕР»РЅРёС‚РµР»СЊРЅРѕРµ РѕРіСЂР°РЅРёС‡РµРЅРёРµ РґРѕСЃС‚СѓРїР° Рє С„Р°Р№Р»Сѓ РЅР° Linux.
    @chmod($target, 0640);

    // РћС‚РЅРѕСЃРёС‚РµР»СЊРЅС‹Р№ РїСѓС‚СЊ Р·Р°РїРёСЃС‹РІР°РµС‚СЃСЏ РІ MySQL.
    return [
        'relative_path' => 'audio/' . $filename,
        'absolute_path' => $target,
        'mime_type' => match ($extension) {
            'mp3' => 'audio/mpeg',
            'wav' => 'audio/wav',
            'ogg' => 'audio/ogg',
        },
        'size_bytes' => $upload['size_bytes'],
    ];
}

/**
 * РЎРѕС…СЂР°РЅСЏРµС‚ РѕР±Р»РѕР¶РєСѓ.
 *
 * Р Р°Р·СЂРµС€РµРЅС‹ JPEG, PNG Рё WebP.
 * SVG РЅР°РјРµСЂРµРЅРЅРѕ РЅРµ СЂР°Р·СЂРµС€С‘РЅ.
 */
function storeCoverUpload(?array $file): ?array
{
    // РћР±Р»РѕР¶РєР° РЅРµРѕР±СЏР·Р°С‚РµР»СЊРЅР°.
    if (
        $file === null
        || (int) ($file['error'] ?? -1) === UPLOAD_ERR_NO_FILE
    ) {
        return null;
    }

    $upload = validateIncomingUpload(
        $file,
        5 * 1024 * 1024,
        'РѕР±Р»РѕР¶РєРё'
    );

    $extension = strtolower(
        pathinfo(
            $upload['original_name'],
            PATHINFO_EXTENSION
        )
    );

    $allowed = [
        'jpg' => 'image/jpeg',
        'jpeg' => 'image/jpeg',
        'png' => 'image/png',
        'webp' => 'image/webp',
    ];

    if (!isset($allowed[$extension])) {
        throw new UploadValidationException(
            'РћР±Р»РѕР¶РєР° РґРѕР»Р¶РЅР° РёРјРµС‚СЊ С„РѕСЂРјР°С‚ JPEG, PNG РёР»Рё WebP.'
        );
    }

    $detectedMime = detectMimeType($upload['tmp_name']);
    $imageInfo = @getimagesize($upload['tmp_name']);

    if (
        $detectedMime !== $allowed[$extension]
        || $imageInfo === false
        || ($imageInfo['mime'] ?? '') !== $allowed[$extension]
    ) {
        throw new UploadValidationException(
            'Р¤Р°Р№Р» РѕР±Р»РѕР¶РєРё РЅРµ РїСЂРѕС€С‘Р» РїСЂРѕРІРµСЂРєСѓ РёР·РѕР±СЂР°Р¶РµРЅРёСЏ.'
        );
    }

    // РћРіСЂР°РЅРёС‡РёРІР°РµРј СЂР°Р·РјРµСЂС‹ РёР·РѕР±СЂР°Р¶РµРЅРёСЏ Рё С‡РёСЃР»Рѕ РїРёРєСЃРµР»РµР№.
    $width = (int) $imageInfo[0];
    $height = (int) $imageInfo[1];

    if (
        $width < 1
        || $height < 1
        || $width > 8000
        || $height > 8000
        || $width * $height > 25000000
    ) {
        throw new UploadValidationException(
            'РќРµРґРѕРїСѓСЃС‚РёРјС‹Рµ СЂР°Р·РјРµСЂС‹ РёР·РѕР±СЂР°Р¶РµРЅРёСЏ.'
        );
    }

    ensurePrivateDirectory(COVER_STORAGE_DIR);

    $filename = bin2hex(random_bytes(16))
        . '.'
        . $extension;

    $target = COVER_STORAGE_DIR
        . DIRECTORY_SEPARATOR
        . $filename;

    if (!move_uploaded_file(
        $upload['tmp_name'],
        $target
    )) {
        throw new RuntimeException(
            'РќРµ СѓРґР°Р»РѕСЃСЊ СЃРѕС…СЂР°РЅРёС‚СЊ РѕР±Р»РѕР¶РєСѓ.'
        );
    }

    @chmod($target, 0640);

    return [
        'relative_path' => 'covers/' . $filename,
        'absolute_path' => $target,
        'mime_type' => $allowed[$extension],
        'size_bytes' => $upload['size_bytes'],
    ];
}

/**
 * РЈРґР°Р»СЏРµС‚ СѓР¶Рµ СЃРѕС…СЂР°РЅС‘РЅРЅС‹Р№ С„Р°Р№Р» РїСЂРё РѕС€РёР±РєРµ РїРѕСЃР»РµРґСѓСЋС‰РµР№ РѕРїРµСЂР°С†РёРё.
 */
function removeStoredUpload(?array $file): void
{
    if (
        $file !== null
        && isset($file['absolute_path'])
        && is_file($file['absolute_path'])
    ) {
        if (!@unlink($file['absolute_path'])) {
            error_log(
                'РќРµ СѓРґР°Р»РѕСЃСЊ СѓРґР°Р»РёС‚СЊ РІСЂРµРјРµРЅРЅС‹Р№ С„Р°Р№Р» Р·Р°РіСЂСѓР·РєРё.'
            );
        }
    }
}

function appConfig(): array
{
    static $config = null;

    if ($config === null) {
        $configFile = __DIR__ . '/config.local.php';

        if (!is_file($configFile)) {
            throw new RuntimeException(
                'РќРµ РЅР°Р№РґРµРЅ С„Р°Р№Р» РєРѕРЅС„РёРіСѓСЂР°С†РёРё РїСЂРёР»РѕР¶РµРЅРёСЏ.'

        $config = require $configFile;
    }

    return $config;
}

/**
 * Р’РѕР·РІСЂР°С‰Р°РµС‚ РїРѕРґРєР»СЋС‡РµРЅРёРµ PDO.
 * РџРѕРґРєР»СЋС‡РµРЅРёРµ СЃРѕР·РґР°С‘С‚СЃСЏ С‚РѕР»СЊРєРѕ РѕРґРёРЅ СЂР°Р· Р·Р° Р·Р°РїСЂРѕСЃ.
 */
function appDb(): PDO
{
    static $pdo = null;

    if ($pdo instanceof PDO) {
        return $pdo;
    }

    $config = appConfig()['db'];

    // РџСЂРё РЅРµРѕР±С…РѕРґРёРјРѕСЃС‚Рё РїР°СЂР°РјРµС‚СЂС‹ РјРѕР¶РЅРѕ РїРµСЂРµРѕРїСЂРµРґРµР»РёС‚СЊ
    // РїРµСЂРµРјРµРЅРЅС‹РјРё РѕРєСЂСѓР¶РµРЅРёСЏ СЃРµСЂРІРµСЂР°.
    $dsn = getenv('DB_DSN')
        ?: ($config['dsn'] ?? '');

    $username = getenv('DB_USER')
        ?: ($config['username'] ?? '');

    $envPassword = getenv('DB_PASS');

    $password = $envPassword !== false
        ? $envPassword
        : ($config['password'] ?? '');

    if ($dsn === '' || $username === '') {
        throw new RuntimeException(
            'РќРµ РЅР°СЃС‚СЂРѕРµРЅС‹ РїР°СЂР°РјРµС‚СЂС‹ РїРѕРґРєР»СЋС‡РµРЅРёСЏ Рє Р‘Р”.'
        );
    }

    $pdo = new PDO($dsn, $username, $password, [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        PDO::ATTR_EMULATE_PREPARES => false,
    ]);

    return $pdo;
}

/**
 * Р’РѕР·РІСЂР°С‰Р°РµС‚ С‚РµРєСѓС‰РµРіРѕ Р°РєС‚РёРІРЅРѕРіРѕ РїРѕР»СЊР·РѕРІР°С‚РµР»СЏ.
 * Р РѕР»СЊ РїСЂРѕРІРµСЂСЏРµС‚СЃСЏ РїРѕ Р‘Р”, Р° РЅРµ С‚РѕР»СЊРєРѕ РїРѕ РґР°РЅРЅС‹Рј СЃРµСЃСЃРёРё.
 */
function requireAuthenticatedUser(PDO $pdo): array
{
    $userId = filter_var(
        $_SESSION['user_id'] ?? null,
        FILTER_VALIDATE_INT,
        ['options' => ['min_range' => 1]]
    );

    if ($userId === false) {
        sendPlainError(401, 'РќРµРѕР±С…РѕРґРёРјР° Р°РІС‚РѕСЂРёР·Р°С†РёСЏ.');
    }

    $stmt = $pdo->prepare(
        'SELECT id, role, is_active
         FROM users
         WHERE id = :id
         LIMIT 1'
    );

    $stmt->execute(['id' => $userId]);
    $user = $stmt->fetch();

    if (!$user || (int) $user['is_active'] !== 1) {
        $_SESSION = [];
        session_regenerate_id(true);

        sendPlainError(
            401,
            'РЈС‡С‘С‚РЅР°СЏ Р·Р°РїРёСЃСЊ РЅРµР°РєС‚РёРІРЅР° РёР»Рё РЅРµ РЅР°Р№РґРµРЅР°.'
        );
    }

    return $user;
}

/**
 * Р”РѕРїСѓСЃРєР°РµС‚ С‚РѕР»СЊРєРѕ СЂРµРґР°РєС‚РѕСЂР° Рё Р°РґРјРёРЅРёСЃС‚СЂР°С‚РѕСЂР°.
 */
function requireEditorOrAdmin(array $user): void
{
    if (!in_array(
        $user['role'],
        ['editor', 'admin'],
        true
    )) {
        sendPlainError(
            403,
            'РќРµРґРѕСЃС‚Р°С‚РѕС‡РЅРѕ РїСЂР°РІ РґР»СЏ РІС‹РїРѕР»РЅРµРЅРёСЏ РѕРїРµСЂР°С†РёРё.'
        );
    }
}

/**
 * Р—Р°РІРµСЂС€Р°РµС‚ Р·Р°РїСЂРѕСЃ СЃ Р±РµР·РѕРїР°СЃРЅС‹Рј С‚РµРєСЃС‚РѕРІС‹Рј СЃРѕРѕР±С‰РµРЅРёРµРј.
 */
function sendPlainError(int $status, string $message): void
{
    http_response_code($status);
    header('Content-Type: text/plain; charset=utf-8');

    exit($message);
}

async function loadCatalog(query = '', tag = '') {
    const grid = document.getElementById('jamendoTrackGrid');
    const localGrid = document.getElementById('localTrackGrid');
    const localCount = Number(grid.dataset.localCount || 0);
    const titleEl = document.getElementById('catalogTitle');
    const countEl = document.getElementById('resultCount');

    isFavoritesView = false;
    if (localGrid) localGrid.classList.remove('hidden');

    grid.innerHTML = `
        <div class="empty-state" style="grid-column: 1 / -1;">

        </div>
    `;

    try {
        const params = new URLSearchParams();
        if (query) params.set('q', query);
        if (tag) params.set('tag', tag);

        const response = await fetch('/api.php?' + params.toString());
        const data = await response.json();
        const tracks = data.tracks || [];

        if (query && tag) {
            titleEl.textContent = `РџРѕРёСЃРє В«${query}В» РїРѕ #${tag}`;

        } else if (tag) {

        } else {
            titleEl.textContent = 'РћС‚РєСЂРѕР№ РґР»СЏ СЃРµР±СЏ';
        }

        countEl.textContent = `${localCount + tracks.length} РјР°С‚РµСЂРёР°Р»РѕРІ`;
        renderTracks(tracks, grid);
    } catch (err) {

        container.appendChild(card);

// 3. РЈРїСЂР°РІР»РµРЅРёРµ РїР»РµРµСЂРѕРј
function playTrackAtIndex(index) {
    if (index < 0 || index >= currentQueue.length) return;
    currentIndex = index;
    currentTrack = currentQueue[index];

    document.body.classList.add('has-player');
    document.getElementById('bottomPlayer').classList.remove('hidden');

    document.getElementById('playerCover').src = currentTrack.cover || '';
    document.getElementById('playerTitle').textContent = currentTrack.title;
    document.getElementById('playerAuthor').textContent = currentTrack.author;

    const dlBtn = document.getElementById('playerDownloadBtn');
    if (currentTrack.canDownload && currentTrack.download) {
        dlBtn.href = currentTrack.download;
        dlBtn.classList.remove('hidden');
    } else {
        dlBtn.classList.add('hidden');
    }

    updatePlayerFavButton();
    audio.src = currentTrack.audio;
    audio.play().catch(() => {});
}

function playNext() {
    if (!currentQueue.length) return;
    playTrackAtIndex((currentIndex + 1) % currentQueue.length);
}

function playPrev() {
    if (!currentQueue.length) return;
    playTrackAtIndex((currentIndex - 1 + currentQueue.length) % currentQueue.length);
}

        document.getElementById('timeCurrent').textContent = formatTime(cur);
        document.getElementById('timeTotal').textContent = formatTime(dur);
    }
});

    document.querySelectorAll('#filterBar .category-chip').forEach(chip => {
        chip.classList.remove('active');

function selectTag(tag) {
    activeTag = tag;
    setActiveChip(`#filterBar [data-tag="${tag}"]`);
    const query = document.getElementById('q').value.trim();
    loadCatalog(query, activeTag);
}

function showFavorites() {
    isFavoritesView = true;
    activeTag = '';
    setActiveChip('#filterBar [data-action="favorites"]');

    const localGrid = document.getElementById('localTrackGrid');
    if (localGrid) localGrid.classList.add('hidden');

    const grid = document.getElementById('jamendoTrackGrid');
    document.getElementById('catalogTitle').textContent = 'РР·Р±СЂР°РЅРЅРѕРµ';
    document.getElementById('resultCount').textContent = `${favorites.length} РјР°С‚РµСЂРёР°Р»РѕРІ`;

    renderTracks(favorites, grid);
}

function toggleFavorite(track) {
    const idx = favorites.findIndex(f => f.id === track.id);
    if (idx === -1) {
        favorites.unshift(track);
    } else {
        favorites.splice(idx, 1);
    }
    localStorage.setItem('mediateka_favs_v2', JSON.stringify(favorites));
    updateFavCounter();
    updatePlayerFavButton();
}

function updateFavCounter() {
    const el = document.getElementById('favCount');
    if (el) el.textContent = favorites.length;
}

function updatePlayerFavButton() {
    if (!currentTrack) return;
    const btn = document.getElementById('playerFavBtn');
    const isLiked = favorites.some(f => f.id === currentTrack.id);
    btn.textContent = isLiked ? 'в™Ґ' : 'в™Ў';
    btn.classList.toggle('liked', isLiked);
}

function formatTime(sec) {
    const total = Math.max(0, Math.floor(Number(sec) || 0));
    const m = Math.floor(total / 60);
    const s = total % 60;
    return `${m}:${s < 10 ? '0' : ''}${s}`;
}

function escapeHtml(str) {
    const div = document.createElement('div');
    div.textContent = str || '';
    return div.innerHTML;
}
